        static int PocetHodu(int[] pole, int hledaneCislo)
        {
            int pocet = 0;

            foreach (int x in pole)
            {
                if (x == hledaneCislo)
                {
                    pocet++;

                }

            }
            return pocet;
        }

        Console.WriteLine("Celkový čas: " + soucet + " minut.");
        Console.WriteLine("Průměrná doba cesty: " + prumer + " minut.");


        static double VypocitejSoucet(double[] pole)
        {
            double soucet = 0;

            foreach (double cas in pole)
            {
                soucet = soucet + cas;
            }

            return soucet;
        }

        static double VypocitejPrumer(double[] pole)
        {
            double soucet = VypocitejSoucet(pole);

            return soucet / pole.Length;



        }
